# Cron jobs for the OAU Mathematics website.
#
# Install:
#   sudo crontab -u www-data -e
#   # Paste the lines below (without the leading # comment markers)
#
# All scripts use --apply to actually execute changes.
# Without --apply they do a dry run (safe to test).
#
# Environment — set at the top of the crontab:
# SHELL=/bin/bash
# PATH=/usr/local/bin:/usr/bin:/bin
# NODE_ENV=production
# DATABASE_URL=postgresql://...
# PRIVATE_STORAGE_DIR=/var/private-storage
# BACKUP_DIR=/var/backups/maths-oau
# RESEND_API_KEY=re_...
# EMAIL_FROM=noreply@maths.oauife.edu.ng

# -----------------------------------------------------------------------
# Backup — daily at 03:00 WAT (02:00 UTC)
# -----------------------------------------------------------------------
# 0 2 * * * /opt/maths-oau/deploy/backup-scripts/daily-backup.sh >> /var/log/maths-oau/backup.log 2>&1

# -----------------------------------------------------------------------
# Alumni cleanup — daily at 04:00 WAT (03:00 UTC)
# Remove orphaned Cloudinary assets for deleted/rejected alumni entries
# -----------------------------------------------------------------------
# 0 3 * * * cd /opt/maths-oau && /usr/bin/npx tsx scripts/alumni-cleanup-orphans.ts --apply >> /var/log/maths-oau/cron.log 2>&1

# -----------------------------------------------------------------------
# Alumni purge rejected — daily at 04:15 WAT (03:15 UTC)
# Permanently delete rejected alumni entries older than 90 days
# -----------------------------------------------------------------------
# 15 3 * * * cd /opt/maths-oau && /usr/bin/npx tsx scripts/alumni-purge-rejected.ts --apply >> /var/log/maths-oau/cron.log 2>&1

# -----------------------------------------------------------------------
# Contact purge archived — daily at 04:30 WAT (03:30 UTC)
# Permanently delete archived contact messages older than 12 months
# -----------------------------------------------------------------------
# 30 3 * * * cd /opt/maths-oau && /usr/bin/npx tsx scripts/contact-purge-archived.ts --apply >> /var/log/maths-oau/cron.log 2>&1

# -----------------------------------------------------------------------
# Journal: purge incomplete drafts — daily at 04:45 WAT (03:45 UTC)
# Delete INCOMPLETE manuscripts inactive for 60+ days (Decision #61)
# -----------------------------------------------------------------------
# 45 3 * * * cd /opt/maths-oau && /usr/bin/npx tsx scripts/journal-purge-drafts.ts --apply >> /var/log/maths-oau/cron.log 2>&1

# -----------------------------------------------------------------------
# Journal: publish scheduled articles — every 15 minutes
# Publishes articles whose scheduledFor date has passed
# -----------------------------------------------------------------------
# */15 * * * * cd /opt/maths-oau && /usr/bin/npx tsx scripts/journal-publish-scheduled.ts --apply >> /var/log/maths-oau/cron.log 2>&1

# -----------------------------------------------------------------------
# Journal: review reminders — daily at 09:00 WAT (08:00 UTC)
# Sends DUE_SOON (3-day warning) and OVERDUE notices to reviewers
# -----------------------------------------------------------------------
# 0 8 * * * cd /opt/maths-oau && /usr/bin/npx tsx scripts/journal-review-reminders.ts --apply >> /var/log/maths-oau/cron.log 2>&1

# -----------------------------------------------------------------------
# Private storage: orphan cleanup — weekly on Sunday at 05:00 WAT (04:00 UTC)
# Removes files in PRIVATE_STORAGE_DIR with no matching DB record
# -----------------------------------------------------------------------
# 0 4 * * 0 cd /opt/maths-oau && /usr/bin/npx tsx scripts/private-storage-orphans.ts --apply >> /var/log/maths-oau/cron.log 2>&1
